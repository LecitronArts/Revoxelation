Output: 4 cascade depth maps rendered per frame, shadow sampling in fragment shaders, PCF soft edges, cascade blending, egui controls for CSM parameters.

  ```glsl
  // Select cascade index from fragment depth in view space
  uint select_cascade(float view_depth, vec4 cascade_splits) { ... }

  // PCF shadow sampling (3x3 kernel)
  float shadow_sample_pcf(sampler2DShadow shadow_map, vec3 shadow_coord, float texel_size) {
      float shadow = 0.0;
      for (int x = -1; x <= 1; x++) {
          for (int y = -1; y <= 1; y++) {
              vec3 offset_coord = shadow_coord + vec3(vec2(x, y) * texel_size, 0.0);
              shadow += texture(shadow_map, offset_coord);
          }
      }
      return shadow / 9.0;
  }

  // Cascade blending (10% transition zone)
  float sample_shadow_csm(sampler2DArrayShadow csm, LightingParams lp, vec3 world_pos, float view_depth) {
      uint cascade = select_cascade(view_depth, lp.cascade_splits);
      vec4 shadow_coord = lp.shadow_matrices[cascade] * vec4(world_pos, 1.0);
      // Perspective divide, bias, PCF sample
      // At cascade boundary: blend with next cascade for flicker-free transition
  }
  ```
